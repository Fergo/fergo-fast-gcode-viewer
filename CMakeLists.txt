cmake_minimum_required(VERSION 3.20)
project(GCodeViewer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL3 CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
add_executable(gcode-viewer src/main.cpp src/gcode.cpp)
set_target_properties(gcode-viewer PROPERTIES WIN32_EXECUTABLE TRUE)
target_link_libraries(gcode-viewer PRIVATE SDL3::SDL3 OpenGL::GL Threads::Threads)
if(WIN32)
  add_custom_command(TARGET gcode-viewer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:gcode-viewer>)
endif()
if(MINGW)
  target_link_options(gcode-viewer PRIVATE -static-libgcc -static-libstdc++)
endif()
add_executable(parser-tests tests/parser_tests.cpp src/gcode.cpp)
target_include_directories(parser-tests PRIVATE src)
enable_testing()
add_test(NAME parser COMMAND parser-tests)
